\chapter*{Resumo}
\addcontentsline{toc}{chapter}{Resumo}

\nota{Resumo entre 250 e 500 palavras. Adaptar do abstract em inglês finalizado para o Congresso de Genética 2026 (Área 7 - Bioinformática Aplicada às Ômicas), expandindo para o escopo maior da tese e traduzindo com precisão os termos técnicos.}

\escrever{Estrutura recomendada:}
\begin{itemize}
    \item \textbf{Contexto} (aprox. 3-4 frases): contaminação por poluentes prioritários; frameworks regulatórios internacionais; fragmentação de evidências funcionais, toxicológicas e regulatórias.
    \item \textbf{Objetivo} (1-2 frases): apresentar o \biorempp{} como ecossistema integrado de quatro componentes.
    \item \textbf{Metodologia} (aprox. 4-5 frases): banco integrado consolidando quatro fontes (\gls{kegg}, \gls{hadeg}, \gls{toxcsm}, curadoria interna); pipeline Snakemake com validação dupla (KEGG-anchored + \gls{gx}); \gls{dbx} para navegação; \gls{bws} para upload de análise; \gls{cli} para pipelines automatizados.
    \item \textbf{Resultados} (aprox. 4-5 frases com números concretos): 384 compostos, 12 classes químicas, 1.543 \glspl{ko}, 994 números \gls{ec}, 1.939 identificadores de reação KEGG, 206 famílias enzimáticas curadas, nove códigos de agências regulatórias (\gls{iarc}, \gls{epa}, \gls{atsdr}, \gls{wfd}, \gls{psl}, \gls{epc}, \gls{conama}); 370 compostos com cobertura toxicológica em 31 endpoints.
    \item \textbf{Conclusão} (2-3 frases): o ecossistema torna evidência integrada acessível para diferentes perfis de usuário e diferentes tipos de dados ômicos, apoiando triagem in silico prévia a estudos experimentais.
\end{itemize}

\vspace{1em}
\textbf{Palavras-chave:} bioinformática; biorremediação; genômica funcional; integração multiômica; compostos prioritários; ecossistema de ferramentas.
